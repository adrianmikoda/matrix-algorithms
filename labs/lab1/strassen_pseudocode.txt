Dense strassenRec(View A, View B) {
    if (A.rows == 2 && A.cols == 2 && B.rows == 2 && B.cols == 2)
        return strassen2x2(A, B);  // 7 mnozen, 18 dodawan/odejmowan

    if (A.rows <= 2 || A.cols <= 2 || B.rows <= 2 || B.cols <= 2)
        return brutal(A, B);

    int h = (maxDimension(A, B) + 1) / 2;  // podzial w polowie: bloki ceil(n/2) i floor(n/2)
    auto [A11, A12, A21, A22] = split2Clipped(A, h);  // brakujace elementy bloku = 0
    auto [B11, B12, B21, B22] = split2Clipped(B, h);

    Dense M1 = strassenRec(A11 + A22, B11 + B22);
    Dense M2 = strassenRec(A21 + A22, B11);
    Dense M3 = strassenRec(A11, B12 - B22);
    Dense M4 = strassenRec(A22, B21 - B11);
    Dense M5 = strassenRec(A11 + A12, B22);
    Dense M6 = strassenRec(A21 - A11, B11 + B12);
    Dense M7 = strassenRec(A12 - A22, B21 + B22);

    Dense C11 = M1 + M4 - M5 + M7;
    Dense C12 = M3 + M5;
    Dense C21 = M2 + M4;
    Dense C22 = M1 - M2 + M3 + M6;

    Dense C(A.rows, B.cols);
    place(C, C11, 0, 0); place(C, C12, 0, h);
    place(C, C21, h, 0); place(C, C22, h, h);
    return C;
}

Matrix strassen(Matrix A, Matrix B) {
    return strassenRec(View(A), View(B)).d;
}
